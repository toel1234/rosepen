<tasks>

<task type="auto">
  <name>Task 1: Clone Inkscape at pinned tag</name>
  <files>src/</files>
  <action>
    Find the latest stable release tag (git ls-remote --tags https://gitlab.com/inkscape/inkscape).
    Clone with submodules (git clone --recurse-submodules --branch TAG).
    Keep remote named `upstream`. Record tag and commit hash in docs/BUILD-NOTES.md.
    Avoid: tracking master, since unstable HEAD breaks a baseline.
  </action>
  <verify>git -C src describe --tags prints the tag; git -C src submodule status shows no '-' prefixes</verify>
  <done>AC-1 satisfied</done>
</task>

<task type="auto">
  <name>Task 2: Install deps and build</name>
  <files>build/, docs/BUILD-NOTES.md</files>
  <action>
    Use Inkscape's documented Linux build deps for the pinned tag (apt build-dep inkscape plus any extras from its INSTALL/doc).
    sudo apt install needs user approval; list packages in BUILD-NOTES.md first and run only with approval.
    cmake -S src -B build -G Ninja -DCMAKE_BUILD_TYPE=Release; cmake --build build -j$(nproc).
    Record missing-package fixes in BUILD-NOTES.md.
  </action>
  <verify>cmake --build build exits 0; ls build/bin/inkscape</verify>
  <done>AC-2 satisfied</done>
</task>

<task type="auto">
  <name>Task 3: Smoke-test binary</name>
  <files>docs/BUILD-NOTES.md</files>
  <action>
    Run build/bin/inkscape --version. Export a small SVG to PNG headless (--export-type=png).
    Note any runtime data-dir env vars needed (INKSCAPE_DATADIR etc.) in BUILD-NOTES.md.
  </action>
  <verify>PNG exists and size > 0</verify>
  <done>AC-3 satisfied</done>
</task>

</tasks>
